날벌레 출현율 GRU

최근 기온·습도·강우·조도·시각의 흐름을 보고 지금 날벌레가 얼마나 나올지(출현율 0~100%) 계산한 뒤, 395nm UV-A LED를 고출력·저출력·소등 3단계로 정하는 알고리즘입니다.

입력 6개 × 24시간 GRU 은닉 상태 16칸 가중치 1,121개 출력 출현율 0~1

작동 흐름

인지

아두이노 나노가 온습도·강우·조도 센서와 RTC 시각을 100ms마다 노트북으로 보냅니다. 노트북은 1분 평균을 냅니다.

판단

비가 오거나 19:00~23:59 밖이면 바로 소등. 나머지 경우 GRU가 최근 24시간 흐름으로 출현율을 계산합니다.

행동

출현율 40% 이상 고출력, 10~40% 저출력, 10% 미만 소등. 아두이노는 비가 감지되면 명령과 상관없이 먼저 끕니다.

직접 돌려보기

문헌 기반 가상 데이터로 학습한 GRU 가중치를 이 페이지 안에서 그대로 계산합니다. 조건을 바꾸면 최근 24시간 날씨를 만들어 넣고 출현율을 다시 계산합니다.

0%소등

GRU 출현율비운영 시간 19~24시

학습 데이터는 실측이 아니라 논문 경향(습도 70% 이상 감소, 일몰 후 0~3시간 집중, 비 그친 직후 비행 중단 등)으로 만든 가상 데이터입니다. 실제 하천 성능은 현장 트랩 개체 수로 다시 학습해 확인합니다.

검증 결과 (가상 2년차 데이터)

1년차(5~9월)로 학습하고 처음 보는 2년차로 평가했습니다. RMSE는 평균적으로 틀린 정도라서 낮을수록 좋습니다.

코드

아두이노는 센서 전송과 안전 규칙, 노트북은 GRU 판단을 맡습니다. 핀 번호와 센서 종류는 실제 회로에 맞게 바꿔 쓰세요.

// 엄물포 | 기상 반응형 395nm 날벌레 유인 장치 (아두이노 나노)
// 역할: 센서 값 전송(인지) + 강우 안전 규칙 + LED 출력(행동)
#include <Wire.h>
#include <RTClib.h>     // RTC 모듈 (DS3231 기준)
#include <DHT.h>        // 온습도 센서 (DHT22 기준)

#define DHT_PIN   2
#define RAIN_PIN  3     // 강우 센서 DO: 젖으면 LOW
#define LUX_PIN   A0    // 조도 센서(CdS): 밝을수록 값이 커지게 배선
#define LED_PIN   9     // MOSFET 게이트 → 395nm LED (PWM)

DHT dht(DHT_PIN, DHT22);
RTC_DS3231 rtc;

int level = 0;                       // 파이썬이 보낸 단계: 0 소등, 1 저출력, 2 고출력
float temp = NAN, hum = NAN;
unsigned long lastSend = 0, lastDht = 0;

void setup() {
  Serial.begin(9600);
  dht.begin();
  rtc.begin();
  pinMode(RAIN_PIN, INPUT);
  pinMode(LED_PIN, OUTPUT);
}

void loop() {
  bool raining = digitalRead(RAIN_PIN) == LOW;

  // 1) 파이썬(GRU)에서 온 명령 읽기: '0' / '1' / '2'
  while (Serial.available()) {
    char c = Serial.read();
    if (c >= '0' && c <= '2') level = c - '0';
  }

  // 2) 온습도는 2초마다만 읽기 (DHT22는 너무 자주 읽으면 오류)
  if (millis() - lastDht >= 2000) {
    lastDht = millis();
    temp = dht.readTemperature();
    hum  = dht.readHumidity();
  }

  // 3) 100ms마다 센서 값 전송: 기온,습도,강우,조도,시
  if (millis() - lastSend >= 100) {
    lastSend = millis();
    DateTime now = rtc.now();
    Serial.print(temp);  Serial.print(',');
    Serial.print(hum);   Serial.print(',');
    Serial.print(raining ? 1 : 0); Serial.print(',');
    Serial.print(analogRead(LUX_PIN)); Serial.print(',');
    Serial.println(now.hour());
  }

  // 4) 행동: 비가 오면 AI 명령과 상관없이 소등 (안전 규칙 최우선)
  int pwm = 0;
  if (!raining) {
    if (level == 2) pwm = 255;       // 고출력
    else if (level == 1) pwm = 100;  // 저출력
  }
  analogWrite(LED_PIN, pwm);
}
"""엄물포 | 실시간 판단 루프 (노트북)
아두이노 → 센서 값 → 1분 평균 → GRU 출현율 → 3단계 → 아두이노
"""
import time
from collections import deque
import numpy as np
import serial
from tensorflow import keras

PORT, BAUD = "COM3", 9600          # 장치 관리자에서 포트 확인
HIGH, LOW = 0.40, 0.10             # 출현율 기준: 40% 이상 고출력, 10% 이상 저출력

model = keras.models.load_model("gru_bug.keras")
ser = serial.Serial(PORT, BAUD, timeout=1)
hours = deque(maxlen=24)           # 최근 24시간 (1시간 평균값)
this_hour = []                     # 이번 시간의 1분 평균값들


def read_minute():
    """1분 동안 들어온 값을 평균. 고장 값(nan)은 버림"""
    ser.reset_input_buffer()       # 쌓여 있던 옛 데이터 버리기 (3~5초 지연 해결)
    rows, end = [], time.time() + 60
    while time.time() < end:
        try:
            t, h, rain, lux, hr = map(float, ser.readline().decode().strip().split(","))
        except ValueError:
            continue
        if np.isnan(t) or np.isnan(h):
            continue
        rows.append([t, h, rain, lux / 1023, hr])
    return np.mean(rows, axis=0) if rows else None


def features(t, h, rain, lux, hr):
    return [(t - 22) / 6, (h - 70) / 15, rain, lux,
            np.sin(2 * np.pi * hr / 24), np.cos(2 * np.pi * hr / 24)]


while True:
    m = read_minute()
    if m is None:                                   # 센서 이상 → 대체 모드
        ser.write(b"1")                             # 기존 조건문처럼 저출력 유지
        continue
    t, h, rain, lux, hr = m
    this_hour.append(features(t, h, rain >= 0.5, lux, round(hr)))
    cur = np.mean(this_hour, axis=0)                # 이번 시간 평균(진행 중)
    recent = (list(hours) + [cur])[-24:]            # 최근 24시간
    if len(this_hour) == 60:                        # 1시간이 차면 기록
        hours.append(cur)
        this_hour = []

    # 판단 1: 안전 규칙이 AI보다 먼저
    in_time = 19 <= round(hr) <= 23
    if rain >= 0.5 or not in_time:
        level = 0
    elif len(recent) < 24:                          # 기록이 쌓이기 전에는 조건문
        level = 2 if h >= 70 else 1
    else:
        # 판단 2: 최근 24시간 → GRU 출현율
        p = float(model.predict(np.array(recent)[None], verbose=0)[0, 0])
        level = 2 if p >= HIGH else 1 if p >= LOW else 0
        print(f"{round(hr)}시  출현율 {p:.0%}  → 단계 {level}")

    ser.write(str(level).encode())                  # 행동: 아두이노로 전송
"""엄물포 | GRU 날벌레 출현율 모델 학습 (TensorFlow/Keras)
virtual_data.csv 열: time, temp, hum, rain, lux, bug_count  (1시간 간격)
 - 지금은 문헌 기반 가상 데이터, 이후 현장 트랩 개체 수로 교체
"""
import numpy as np
import pandas as pd
from tensorflow import keras
from tensorflow.keras import layers

WIN = 24                                   # 최근 24시간을 보고 예측
FEATS = ["temp", "hum", "rain", "lux", "hour_sin", "hour_cos"]

df = pd.read_csv("virtual_data.csv", parse_dates=["time"])
hour = df["time"].dt.hour
df["hour_sin"] = np.sin(2 * np.pi * hour / 24)   # 23시와 0시가 가깝도록 시각을 원 위의 점으로
df["hour_cos"] = np.cos(2 * np.pi * hour / 24)
df["temp"] = (df["temp"] - 22) / 6               # 표준화: 단위가 다른 값을 비슷한 크기로
df["hum"] = (df["hum"] - 70) / 15
df["lux"] = df["lux"].clip(0, 1)                 # 0(어두움) ~ 1(밝음)으로 맞춘 값

# 출현율 0~1: 상위 1% 개체 수를 1로 보고 나눔
y_all = (df["bug_count"] / df["bug_count"].quantile(0.99)).clip(0, 1).values
X_all = df[FEATS].values.astype("float32")

# 슬라이딩 윈도우: [i-23 ... i] 24시간 → i시점 출현율
X = np.stack([X_all[i - WIN + 1:i + 1] for i in range(WIN - 1, len(df))])
y = y_all[WIN - 1:]

# 시간 순서대로 나누기: 앞 절반(1년차) 학습, 뒤 절반(2년차) 평가
split = len(X) // 2
X_tr, y_tr, X_te, y_te = X[:split], y[:split], X[split:], y[split:]

model = keras.Sequential([
    layers.Input(shape=(WIN, len(FEATS))),
    layers.GRU(16),                         # 은닉 상태(기억) 16칸
    layers.Dense(1, activation="sigmoid"),  # 출현율 0~1
])
model.compile(optimizer="adam", loss="mse")
model.fit(X_tr, y_tr, epochs=30, batch_size=32, validation_split=0.1, verbose=2)

pred = model.predict(X_te).ravel()
rmse = np.sqrt(np.mean((pred - y_te) ** 2))
r2 = 1 - np.sum((pred - y_te) ** 2) / np.sum((y_te - y_te.mean()) ** 2)
r = np.corrcoef(pred, y_te)[0, 1]
print(f"RMSE {rmse:.3f}  R² {r2:.3f}  r {r:.3f}")   # 성공 기준: r ≥ 0.7

model.save("gru_bug.keras")
